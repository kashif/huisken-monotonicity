\section{Introduction}

This blueprint documents a Lean 4 formalization of Huisken's monotonicity formula for
mean curvature flow~\cite{huisken1990}, built on mathlib~\cite{mathlib2020} and on the
\texttt{DifferentialGeometry} library~\cite{differentialgeometry}.

Let $F \colon M \times \mathbb{R} \to E$ be a smooth family of immersions of a closed
manifold $M$ of dimension $n$ into a finite-dimensional inner product space $E$, moving
by mean curvature, $\partial_t F = \vec H$. For the backward heat kernel
\[
  \rho(x, t) = (4\pi(t_0 - t))^{-n/2}\,\exp\!\left(-\frac{\|x\|^2}{4(t_0 - t)}\right)
\]
and every $t < t_0$,
\[
  \frac{d}{dt}\int_M \rho(F, t)\,d\mu_t = -\int_M \rho(F, t)\,
    \left\| \vec H + \frac{F^\perp}{2(t_0 - t)} \right\|^2 d\mu_t
\]
(Theorem~\ref{thm:huisken-monotonicity}).

\section*{Formalization status}

The theorem is proved with no hypotheses beyond the flow itself, for an arbitrary centre,
in any codimension and without choosing a unit normal, for flows on a time interval;
together with the monotonicity it implies, Huisken's Corollary 3.2 for the rescaled flow,
and the shrinking sphere as an instance. \texttt{lake build} reports no \texttt{sorry}, and the
theorem depends only on Lean's standard axioms.

\section{The backward heat kernel}

\begin{definition}[backward heat kernel]
  \label{def:backward-heat-kernel}
  \lean{HuiskenMonotonicity.backwardHeatKernel}
  \leanok
  For a hypersurface dimension $n \in \mathbb{N}$, base time $t_0 \in \mathbb{R}$,
  current time $t \in \mathbb{R}$, and ambient point $x$ in a (semi)normed real
  inner product space, the \emph{backward heat kernel} is
  \[
    \rho(x, t) = (4\pi(t_0 - t))^{-n/2}\,
      \exp\!\left(-\frac{\|x\|^2}{4(t_0 - t)}\right).
  \]
\end{definition}

\begin{lemma}[positivity]
  \label{lem:rho-pos}
  \lean{HuiskenMonotonicity.backwardHeatKernel_pos}
  \leanok
  \uses{def:backward-heat-kernel}
  For every $t < t_0$ and every $x$, $\rho(x, t) > 0$.
\end{lemma}
\begin{proof}\leanok Verified in Lean. \end{proof}

\begin{lemma}[time derivative]
  \label{lem:rho-dt}
  \lean{HuiskenMonotonicity.hasDerivAt_backwardHeatKernel_time}
  \leanok
  \uses{def:backward-heat-kernel}
  For every $t < t_0$ and every $x$,
  \[
    \frac{\partial \rho}{\partial t}(x, t) = \rho(x, t)
      \left(\frac{n}{2(t_0 - t)} - \frac{\|x\|^2}{4(t_0 - t)^2}\right).
  \]
\end{lemma}
\begin{proof}\leanok Verified in Lean. \end{proof}

\begin{lemma}[directional spatial derivative]
  \label{lem:rho-dx}
  \lean{HuiskenMonotonicity.hasDerivAt_backwardHeatKernel_dir}
  \leanok
  \uses{def:backward-heat-kernel}
  For every $t < t_0$, every $x, v$, and every $s$,
  \[
    \left.\frac{d}{ds'}\right|_{s' = 0} \rho(x + s' v, t)
      = -\frac{\rho(x, t)}{2(t_0 - t)}\,\langle x, v\rangle.
  \]
  Equivalently, $\nabla_x \rho(x, t) = -\rho(x, t)\,x / (2(t_0 - t))$.
\end{lemma}
\begin{proof}\leanok Verified in Lean. \end{proof}

\begin{lemma}[second directional derivative along a line]
  \label{lem:rho-dxx}
  \lean{HuiskenMonotonicity.iteratedDeriv_two_backwardHeatKernel_along}
  \leanok
  \uses{lem:rho-dx}
  For every $t < t_0$, every $x, v$,
  \[
    \frac{d^2}{ds^2}\bigg|_{s=0} \rho(x + s v, t)
      = \rho(x, t)
        \left(\frac{\langle x, v\rangle^2}{4(t_0 - t)^2}
          - \frac{\|v\|^2}{2(t_0 - t)}\right).
  \]
\end{lemma}
\begin{proof}\leanok Verified in Lean. \end{proof}

\begin{theorem}[backward heat equation in coordinates]
  \label{thm:rho-heat-coord}
  \lean{HuiskenMonotonicity.backwardHeatEquation_euclidean}
  \leanok
  \uses{lem:rho-dt, lem:rho-dxx}
  On $\mathrm{EuclideanSpace}\,\mathbb{R}\,(\mathrm{Fin}\,n)$, the backward heat
  kernel satisfies
  \[
    \frac{\partial \rho}{\partial t}(x, t)
      + \sum_{i=1}^{n} \frac{\partial^2 \rho}{\partial x_i^2}(x, t) = 0
    \qquad \text{for } t < t_0.
  \]
\end{theorem}
\begin{proof}\leanok Verified in Lean. \end{proof}

\begin{theorem}[backward heat equation, Laplacian form]
  \label{thm:rho-heat-laplacian}
  \lean{HuiskenMonotonicity.backwardHeatEquation_laplacian}
  \leanok
  \uses{thm:rho-heat-coord}
  Equivalently, against mathlib's coordinate-free Laplacian $\Delta$,
  $\partial_t \rho + \Delta \rho = 0$ on
  $\mathrm{EuclideanSpace}\,\mathbb{R}\,(\mathrm{Fin}\,n)$.
\end{theorem}
\begin{proof}\leanok Verified in Lean. \end{proof}

\begin{theorem}[total mass]
  \label{thm:rho-integral}
  \lean{HuiskenMonotonicity.integral_backwardHeatKernel}
  \leanok
  \uses{def:backward-heat-kernel}
  For any finite-dimensional real inner product space $V$ with
  $\dim_{\mathbb{R}} V = n$ and any $t < t_0$,
  \[
    \int_V \rho(x, t)\,dx = 1.
  \]
\end{theorem}
\begin{proof}\leanok Verified in Lean. \end{proof}

\section{Immersions and the induced metric}

\begin{definition}[smooth immersion, smooth immersion family]
  \label{def:smooth-immersion-family}
  \lean{HuiskenMonotonicity.SmoothImmersion, HuiskenMonotonicity.SmoothImmersionFamily, HuiskenMonotonicity.SmoothImmersionFamily.slice}
  \leanok
  A smooth map $f \colon M \to E$ with injective differential; and a map
  $F \colon M \times \mathbb{R} \to E$, smooth on the product manifold, every time slice
  $F_t$ of which is an immersion. All fixed-time geometry is attached to a single immersion,
  and a family reads it off its slices.
\end{definition}

\begin{definition}[induced metric and surface measure]
  \label{def:induced-metric}
  \lean{HuiskenMonotonicity.SmoothImmersion.inducedMetric, HuiskenMonotonicity.SmoothImmersion.surfaceMeasure}
  \leanok
  \uses{def:smooth-immersion-family}
  The pullback $g_t(v, w) = \langle dF_t\,v, dF_t\,w\rangle$ of the ambient inner
  product, as a smooth Riemannian metric on $M$, and its Riemannian volume measure
  $\mu_t$.
\end{definition}

\section{Extrinsic calculus}

Everything in this section is stated for a map $F \colon E_1 \to E$ between model spaces,
a frame $b$ of $E_1$ and a matrix $g^{ij}$ of inverse-metric components. On a manifold it
is applied to the chart representation of a time slice.

\begin{theorem}[Weingarten equation, orientation-free]
  \label{thm:weingarten}
  \lean{HuiskenMonotonicity.inner_fderiv_eq_neg_inner_secondDerivative}
  \leanok
  If $\xi$ is normal along $F$ near $x$, then
  $\langle d\xi_x(v), dF_x(w)\rangle = -\langle \xi(x), D^2F_x(v, w)\rangle$.
\end{theorem}

\begin{proof}
  \leanok
  Differentiate $\langle \xi, dF(w)\rangle \equiv 0$ in the direction $v$ and use the product rule.
\end{proof}

\begin{definition}[tangential divergence, normal part, mean curvature vector]
  \label{def:tangential-divergence}
  \lean{HuiskenMonotonicity.tangentialDivergence, HuiskenMonotonicity.tangentialPart, HuiskenMonotonicity.normalPart, HuiskenMonotonicity.meanCurvatureNormal}
  \leanok
  $\operatorname{div}_M \xi = g^{ij}\langle d\xi(b_i), \partial_j F\rangle$;
  $v^\top = g^{kl}\langle v, \partial_l F\rangle\,\partial_k F$ and $v^\perp = v - v^\top$;
  $\vec H = (g^{ij} D^2F(b_i, b_j))^\perp$.
\end{definition}

\begin{theorem}[normal divergence identity]
  \label{thm:normal-divergence}
  \lean{HuiskenMonotonicity.tangentialDivergence_of_isNormalAlong}
  \leanok
  \uses{thm:weingarten, def:tangential-divergence}
  For $\xi$ normal along $F$, $\operatorname{div}_M \xi = -\langle \xi, \vec H\rangle$.
\end{theorem}

\begin{proof}
  \leanok
  Contract Theorem~\ref{thm:weingarten} with $g^{ij}$.
\end{proof}

\begin{theorem}[$\operatorname{div}_M F = n$]
  \label{thm:div-position}
  \lean{HuiskenMonotonicity.tangentialDivergence_self}
  \leanok
  \uses{def:tangential-divergence}
  The tangential divergence of the position vector is the dimension.
\end{theorem}

\begin{proof}
  \leanok
  $\operatorname{div}_M F = g^{ij} g_{ij} = n$.
\end{proof}

\begin{theorem}[Beltrami's formula for $|F|^2/2$]
  \label{thm:beltrami-frame}
  \lean{HuiskenMonotonicity.tangentialDivergence_tangentialPart_position}
  \leanok
  \uses{thm:normal-divergence, thm:div-position}
  $\operatorname{div}_M F^\top = n + \langle F, \vec H\rangle$, by the split
  $F = F^\top + F^\perp$.
\end{theorem}

\begin{proof}
  \leanok
  Split $F^\top = F - F^\perp$: $\operatorname{div}_M F = n$, and $F^\perp$ is normal, so $\operatorname{div}_M F^\perp = -\langle F^\perp, \vec H\rangle = -\langle F, \vec H\rangle$.
\end{proof}

\begin{theorem}[Voss--Weyl]
  \label{thm:voss-weyl}
  \lean{HuiskenMonotonicity.SmoothImmersion.vossWeyl_eq_tangentialDivergence}
  \leanok
  \uses{def:tangential-divergence, def:induced-metric}
  In a chart, the library's Voss--Weyl divergence
  $\frac{1}{\sqrt g}\partial_i(\sqrt g\,Z^i)$ equals the tangential divergence of the
  tangential field $Z^k \partial_k F$.
\end{theorem}

\begin{proof}
  \leanok
  Both sides equal $\partial_i Z^i + Z^i \cdot \tfrac12 g^{kl}\partial_i g_{kl}$: the left by the library's Jacobi formula, the right by the contracted Christoffel identity.
\end{proof}

\begin{theorem}[the mean curvature vector is the library's]
  \label{thm:mean-curvature-bridge}
  \lean{HuiskenMonotonicity.SmoothImmersion.meanCurvatureVectorAt_eq, DifferentialGeometry.Geometry.meanCurvatureVectorAt, DifferentialGeometry.Geometry.secondFundamentalFormAmbientAt_eq_modelValue}
  \leanok
  \uses{def:mean-curvature-M}
  For an immersion $f$ into a Euclidean space, $\vec H$ is the metric trace of the
  \texttt{DifferentialGeometry} second fundamental form of $f$, as a Riemannian isometric
  immersion, read in $E$ through mathlib's identification of $T_{f(p)}E$ with $E$. In charts
  that form is $\partial^2 F + \Gamma_E - dF(\Gamma_M)$; the Euclidean
  $\Gamma_E$ vanishes and $dF(\Gamma_M)$ is tangent.
\end{theorem}

\begin{proof}
  \leanok
  In charts the library's second fundamental form differs from $\partial^2 F$ by $\Gamma_E = 0$ and by the tangent vector $dF(\Gamma_M)$; it is normal, so it is the normal part of $\partial^2 F$. Both traces use the same inverse metric.
\end{proof}

\section{The first variation}

\begin{definition}[mean curvature and normal position on $M$]
  \label{def:mean-curvature-M}
  \lean{HuiskenMonotonicity.SmoothImmersion.meanCurvature, HuiskenMonotonicity.SmoothImmersion.normalPosition, HuiskenMonotonicity.SmoothImmersion.tangentialSqNorm}
  \leanok
  \uses{def:tangential-divergence, def:induced-metric}
  $\vec H$, $F^\perp$ and $\|F^\top\|^2$ at $p \in M$, read in the chart at $p$.
\end{definition}

\begin{theorem}[Beltrami's formula on $M$]
  \label{thm:laplacian-half-sq-norm}
  \lean{HuiskenMonotonicity.SmoothImmersion.laplacian_halfSqNorm}
  \leanok
  \uses{thm:beltrami-frame, thm:voss-weyl, def:mean-curvature-M}
  The library's Laplacian of $u = |F|^2/2$ is $\Delta u = n + \langle F, \vec H\rangle$.
\end{theorem}

\begin{proof}
  \leanok
  The library's Voss--Weyl formula for the Laplacian, Theorem~\ref{thm:voss-weyl}, and Theorem~\ref{thm:beltrami-frame} in the chart at the point.
\end{proof}

\begin{lemma}[derivative of the kernel along $\nabla u$]
  \label{lem:X-rho}
  \lean{HuiskenMonotonicity.SmoothImmersion.tangentSectionAction_gradG_halfSqNorm_backwardHeatKernel}
  \leanok
  \uses{def:mean-curvature-M, def:backward-heat-kernel}
  $\langle \nabla u, \nabla \rho\rangle = -\frac{\rho}{2\tau}\|F^\top\|^2$.
\end{lemma}

\begin{proof}
  \leanok
  The library's chart formula $\langle \nabla u, \nabla \rho\rangle = g^{ij}\partial_i u\,\partial_j \rho$, with $\partial_i u = \langle F, \partial_i F\rangle$ and $\partial_j \rho = -\frac{\rho}{2\tau}\langle F, \partial_j F\rangle$.
\end{proof}

\begin{theorem}[first variation identity]
  \label{thm:first-variation}
  \lean{HuiskenMonotonicity.SmoothImmersion.integral_firstVariation}
  \leanok
  \uses{thm:laplacian-half-sq-norm, lem:X-rho}
  On a closed manifold,
  $\int_M \rho\left(n - \frac{\|F^\top\|^2}{2\tau} + \langle F, \vec H\rangle\right)d\mu_t = 0$,
  by the library's integration by parts $\int X(\rho) = -\int \rho \operatorname{div} X$
  with $X = \nabla u$.
\end{theorem}

\begin{proof}
  \leanok
  Huisken applies the first variation formula to $Y = \rho F/\tau$. Here the integrand is $X(\rho) + \rho \operatorname{div} X$ for $X = \nabla u$, and the library's integration by parts gives $\int X(\rho) = -\int \rho \operatorname{div} X$.
\end{proof}

\section{Mean curvature flow}

\begin{definition}[mean curvature flow]
  \label{def:mcf}
  \lean{HuiskenMonotonicity.SmoothImmersionFamily.IsMeanCurvatureFlowAt}
  \leanok
  \uses{def:mean-curvature-M}
  $\partial_t F(p, t) = \vec H(p, t)$ for all $p$, at the time $t$. The formula at $t$
  needs the flow equation at $t$ only.
\end{definition}

\begin{lemma}[normality in every chart]
  \label{lem:normal-every-chart}
  \lean{HuiskenMonotonicity.SmoothImmersion.inner_meanCurvature_fderiv_chartRep}
  \leanok
  \uses{def:mean-curvature-M}
  $\vec H(p)$ is orthogonal to the tangent vectors of the chart representation of $F_t$ in
  any chart containing $p$.
\end{lemma}

\begin{proof}
  \leanok
  The library's fixed-chart differential reduces the chart tangent vectors to $df$, and $\vec H$ is normal to the range of $df$.
\end{proof}

\begin{theorem}[evolution of the volume element]
  \label{thm:volume-evolution}
  \lean{HuiskenMonotonicity.SmoothImmersionFamily.traceTimeDerivMetric_eq_two_mul_tangentialDivergence, HuiskenMonotonicity.SmoothImmersion.tangentialDivergence_eq_neg_norm_sq_meanCurvature, HuiskenMonotonicity.SmoothImmersionFamily.traceTimeDerivMetric_of_isMeanCurvatureFlowAt}
  \leanok
  \uses{def:mcf, lem:normal-every-chart, thm:normal-divergence}
  For any family, $\operatorname{tr}(g^{-1}\partial_t g) = 2\operatorname{div}_M V$ for the
  velocity $V$ (the first variation of area, $\partial_t\,d\mu = \operatorname{div}_M V\,d\mu$).
  Under mean curvature flow $\operatorname{div}_M \vec H = -\|\vec H\|^2$, so
  $\operatorname{tr}(g^{-1}\partial_t g) = -2\|\vec H\|^2$.
\end{theorem}

\begin{proof}
  \leanok
  $\partial_t g_{ij} = \langle \partial_i V, \partial_j F\rangle + \langle \partial_i F, \partial_j V\rangle$ by the product rule; its trace is $2\operatorname{div}_M V$. For $V = \vec H$, which is normal, Theorem~\ref{thm:normal-divergence} gives $\operatorname{div}_M \vec H = -\langle \vec H, \vec H\rangle$. This is Huisken's $\frac{d}{dt}\,d\mu_t = -H^2\,d\mu_t$.
\end{proof}

\begin{theorem}[transport formula]
  \label{thm:transport}
  \lean{HuiskenMonotonicity.hasDerivAt_integral_backwardHeatKernel}
  \leanok
  \uses{def:induced-metric, def:backward-heat-kernel}
  $\frac{d}{dt}\int \rho\,d\mu_t = \int \left(\partial_t[\rho(F)] +
  \tfrac12 \operatorname{tr}(g^{-1}\partial_t g)\,\rho\right)d\mu_t$, through the
  library's volume variation formula.
\end{theorem}

\begin{proof}
  \leanok
  The library's volume variation formula, with the integrand cut off in time away from the singular time $t_0$.
\end{proof}

\begin{lemma}[the kernel along the flow]
  \label{lem:kernel-curve}
  \lean{HuiskenMonotonicity.hasDerivAt_backwardHeatKernel_curve}
  \leanok
  \uses{lem:rho-dt, lem:rho-dx}
  $\frac{d}{ds}\rho(\gamma(s), s) = \rho\left(\frac{n}{2\tau} - \frac{\|\gamma\|^2}{4\tau^2}
  - \frac{\langle \gamma, \gamma'\rangle}{2\tau}\right)$.
\end{lemma}

\begin{proof}
  \leanok
  The chain rule for the jointly smooth kernel.
\end{proof}

\section{Huisken's monotonicity formula}

\begin{lemma}[pointwise identity]
  \label{lem:pointwise-identity}
  \lean{HuiskenMonotonicity.huisken_pointwise_identity}
  \leanok
  If $F^\top \perp F^\perp$ and $\vec H \perp F^\top$, then
  \[
    \rho\left(\frac{n}{2\tau} - \frac{\|F\|^2}{4\tau^2} - \frac{\langle F, \vec H\rangle}{2\tau}
      - \|\vec H\|^2\right)
    = -\rho\left\|\vec H + \frac{F^\perp}{2\tau}\right\|^2
      + \frac{1}{2\tau}\,\rho\left(n - \frac{\|F^\top\|^2}{2\tau} + \langle \vec H, F\rangle\right).
  \]
\end{lemma}

\begin{proof}
  \leanok
  Expand $\|\vec H + F^\perp/(2\tau)\|^2$ and $\|F\|^2 = \|F^\top\|^2 + \|F^\perp\|^2$.
\end{proof}

\begin{theorem}[Huisken's monotonicity formula]
  \label{thm:huisken-monotonicity}
  \lean{HuiskenMonotonicity.SmoothImmersionFamily.huisken_monotonicity}
  \leanok
  \uses{thm:transport, thm:volume-evolution, lem:kernel-curve, lem:pointwise-identity, thm:first-variation}
  For a family of immersions of a closed manifold moving by mean curvature at time $t$, and
  the kernel centred at $(x_0, t_0)$ with $t < t_0$,
  \[
    \frac{d}{dt}\int_M \rho_{x_0,t_0}(F, t)\,d\mu_t = -\int_M \rho_{x_0,t_0}(F, t)\,
      \left\| \vec H + \frac{(F - x_0)^\perp}{2(t_0 - t)} \right\|^2 d\mu_t.
  \]
  The centre is reduced to the origin by translating the family, which changes neither the
  metric, nor the measure, nor $\vec H$ (Huisken's Remark (i) after Theorem 3.1).
\end{theorem}

\begin{proof}
  \leanok
  As in Huisken: Theorem~\ref{thm:transport} with Theorem~\ref{thm:volume-evolution} and Lemma~\ref{lem:kernel-curve}, then Lemma~\ref{lem:pointwise-identity}; the first-variation term integrates to zero by Theorem~\ref{thm:first-variation}. The centre is reduced to the origin by a translation.
\end{proof}

\section{Flows on a time interval, and monotonicity}

A mean curvature flow of a closed manifold exists only on a bounded time interval.

\begin{definition}[flow on an open set of times]
  \label{def:flow-on}
  \lean{HuiskenMonotonicity.ImmersionFlowOn, HuiskenMonotonicity.ImmersionFlowOn.IsMeanCurvatureFlow}
  \leanok
  \uses{def:smooth-immersion-family, def:mcf}
  A family of immersions smooth on $M \times J$ for an open $J \subseteq \mathbb{R}$, moving
  by mean curvature at every time of $J$.
\end{definition}

\begin{theorem}[Theorem 3.1 on an interval]
  \label{thm:huisken-interval}
  \lean{HuiskenMonotonicity.ImmersionFlowOn.huisken_monotonicity, HuiskenMonotonicity.ImmersionFlowOn.extendAt, HuiskenMonotonicity.ImmersionFlowOn.hasDerivAt_integral_of_extendAt}
  \leanok
  \uses{def:flow-on, thm:huisken-monotonicity}
  The formula holds at every $t \in J$ with $t < t_0$: a smooth reparametrisation of time
  that is the identity near $t$ extends the flow to all times without changing it near $t$.
\end{theorem}

\begin{proof}
  \leanok
  Reparametrise time by a smooth $\varphi \colon \mathbb{R} \to J$ equal to the identity near $t$, and apply Theorem~\ref{thm:huisken-monotonicity} to the extended family. It moves by mean curvature at $t$, and it agrees with the flow, its measure and the weighted integral near $t$, so the derivatives at $t$ coincide. The same transfer gives the rescaled version of Corollary~\ref{cor:rescaled}.
\end{proof}

\begin{corollary}[monotonicity]
  \label{cor:antitone}
  \lean{HuiskenMonotonicity.ImmersionFlowOn.antitoneOn_integral}
  \leanok
  \uses{thm:huisken-interval}
  For an interval $J$, $t \mapsto \int_M \rho_{x_0,t_0}\,d\mu_t$ is non-increasing on
  $J \cap (-\infty, t_0)$.
\end{corollary}

\begin{proof}
  \leanok
  The derivative is $-\int \rho \|\cdot\|^2 \le 0$ on the convex open set $J \cap (-\infty, t_0)$.
\end{proof}

\section{The rescaled flow}

\begin{definition}[rescaled flow, rescaled kernel]
  \label{def:rescaled}
  \lean{HuiskenMonotonicity.SmoothImmersionFamily.IsRescaledMeanCurvatureFlowAt, HuiskenMonotonicity.rescaledHeatKernel}
  \leanok
  $\partial_s \tilde F = \tilde H + \tilde F$ (Huisken's equation (4)), and
  $\tilde\rho(x) = e^{-|x|^2/2}$, the kernel at $\tau = 1/2$ up to a constant.
\end{definition}

\begin{lemma}[volume element under the rescaled flow]
  \label{lem:rescaled-volume}
  \lean{HuiskenMonotonicity.SmoothImmersionFamily.traceTimeDerivMetric_of_isRescaledMeanCurvatureFlowAt, HuiskenMonotonicity.trace_frameGram_deriv_eq_two_mul_tangentialDivergence}
  \leanok
  \uses{def:rescaled, thm:normal-divergence, thm:div-position}
  $\frac{d}{ds}\,d\mu_s = (n - \tilde H^2)\,d\mu_s$: the trace of the metric evolution is
  $2 \operatorname{div}_M(\tilde H + \tilde F) = 2(n - |\tilde H|^2)$.
\end{lemma}

\begin{proof}
  \leanok
  The trace of the metric evolution is $2\operatorname{div}_M(\tilde H + \tilde F) = 2(n - |\tilde H|^2)$, by Theorems~\ref{thm:normal-divergence} and~\ref{thm:div-position}.
\end{proof}

\begin{corollary}[Huisken's Corollary 3.2]
  \label{cor:rescaled}
  \lean{HuiskenMonotonicity.SmoothImmersionFamily.huisken_monotonicity_rescaled, HuiskenMonotonicity.ImmersionFlowOn.huisken_monotonicity_rescaled, HuiskenMonotonicity.ImmersionFlowOn.antitoneOn_integral_rescaled}
  \leanok
  \uses{lem:rescaled-volume, thm:first-variation, lem:pointwise-identity}
  $\frac{d}{ds}\int_M \tilde\rho\,d\mu_s = -\int_M \tilde\rho\,|\tilde H + \tilde F^\perp|^2\,d\mu_s$,
  by the calculation of Theorem 3.1 at $\tau = 1/2$; so $s \mapsto \int_M \tilde\rho\,d\mu_s$
  is non-increasing.
\end{corollary}

\begin{proof}
  \leanok
  Huisken: ``The identity follows from $d\mu_s/ds = (n - H^2)\mu_s$ and a calculation analogous to the proof of Theorem 3.1.'' Since $\tilde\rho$ is the kernel at $\tau = 1/2$ up to a constant, Theorem~\ref{thm:first-variation} and Lemma~\ref{lem:pointwise-identity} apply at $\tau = 1/2$.
\end{proof}

\begin{theorem}[Huisken's rescaling, \S 2]
  \label{thm:rescaling}
  \lean{HuiskenMonotonicity.SmoothImmersion.meanCurvature_dilate, HuiskenMonotonicity.ImmersionFlowOn.rescale, HuiskenMonotonicity.ImmersionFlowOn.IsMeanCurvatureFlow.rescale, HuiskenMonotonicity.ImmersionFlowOn.antitoneOn_integral_rescale}
  \leanok
  \uses{def:flow-on, cor:rescaled}
  For a mean curvature flow on $(a, T)$, the surfaces
  $\tilde F(\cdot, s) = (2(T - t))^{-1/2} F(\cdot, t)$, $s = -\tfrac12 \log(T - t)$, defined
  for $s > -\tfrac12 \log(T - a)$, satisfy $\partial_s \tilde F = \tilde H + \tilde F$
  (Huisken's equation (4)), because $\vec H(cF) = c^{-1} \vec H(F)$. Hence
  $s \mapsto \int_M \tilde\rho\,d\tilde\mu_s$ is non-increasing.
\end{theorem}

\begin{proof}
  \leanok
  The chain rule in time and the product rule with the factor $e^s/\sqrt2$, together with $\vec H(cF) = c^{-1}\vec H(F)$: the metric scales by $c^2$, its inverse by $c^{-2}$, the second derivative by $c$, and the normal projection is unchanged.
\end{proof}

\section{The shrinking sphere}

\begin{lemma}[mean curvature of a sphere]
  \label{lem:sphere-mean-curvature}
  \lean{HuiskenMonotonicity.SmoothImmersion.meanCurvature_eq_of_norm_eq}
  \leanok
  \uses{thm:beltrami-frame}
  If $|f| = R$ and the normal space is spanned by $f$, then $\vec H = -\frac{n}{R^2} f$:
  $f^\top = 0$, so Beltrami's formula gives $\langle f, \vec H\rangle = -n$.
\end{lemma}

\begin{proof}
  \leanok
  $|f|$ is constant, so $f^\top = 0$ and Theorem~\ref{thm:beltrami-frame} gives $0 = n + \langle f, \vec H\rangle$; $\vec H$ is normal, hence a multiple of $f$.
\end{proof}

\begin{theorem}[the shrinking sphere satisfies the hypotheses]
  \label{thm:shrinking-sphere}
  \lean{HuiskenMonotonicity.shrinkingSphere, HuiskenMonotonicity.shrinkingSphere_isMeanCurvatureFlow, HuiskenMonotonicity.shrinkingSphere_antitoneOn_integral}
  \leanok
  \uses{lem:sphere-mean-curvature, cor:antitone}
  $F(p, t) = \sqrt{1 - 2nt}\,p$ on $S^n$, $2nt < 1$, moves by mean curvature, and its
  Gaussian density is non-increasing.
\end{theorem}

\begin{proof}
  \leanok
  By Lemma~\ref{lem:sphere-mean-curvature}, $\vec H = -\frac{n}{R}\,p$ on the sphere of radius $R(t) = \sqrt{1 - 2nt}$, and $\partial_t(R\,p) = -\frac{n}{R}\,p$. Corollary~\ref{cor:antitone} applies.
\end{proof}
